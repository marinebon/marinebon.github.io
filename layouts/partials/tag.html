{{/* Renders a single tag chip. Argument is EITHER: - a "facet.value" string -> a (safe inside card links, no nested ) - a dict {value, link:true} -> an linking to the /tags// term page Splits on the first "." to get facet and value; strips the prefix for display. Falls back to the full string if no "." is found. Matching is case-insensitive so it works both for original-case tags in front matter and the lowercased terms Hugo emits on taxonomy pages. Color comes from .tag--{facet}. The link URL is built from the lowercased tag (Hugo lowercases taxonomy terms) run through relURL so it carries the site base path — checked by scripts/check_links.py. */}} {{ $raw := . }}{{ $link := false }} {{ if reflect.IsMap . }}{{ $raw = .value }}{{ $link = .link }}{{ end }} {{ $parts := split $raw "." }} {{ if ge (len $parts) 2 }} {{ $facet := index $parts 0 }} {{ $value := strings.TrimPrefix (printf "%s." $facet) $raw }} {{/* friendly label from the canonical registry (shared with the grouped lists) */}} {{ $label := partial "tag-label.html" (dict "facet" $facet "value" $value) }} {{ if $link }}{{ $label }} {{ else }}{{ $label }}{{ end }} {{ else }} {{ if $link }}{{ humanize $raw }} {{ else }}{{ humanize $raw }}{{ end }} {{ end }}
